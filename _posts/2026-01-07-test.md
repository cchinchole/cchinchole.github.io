section .data
    hello db "Hello, world!", 0Ah

section .text
    global _start

    ; write the string to stdout
    mov eax, 4
    mov ebx, 1
    mov ecx, hello
    mov edx, hellolen
    int 80h

    ; exit the program
    mov eax, 1
    xor ebx, ebx
    int 80h
